# Istruzioni per LLM
# Feature indica il modulo python da importare il quale contiene le fn:
# Gli Scenario sono i nomi delle funzioni di test che dovrai generare
# Given contiene la descrizione dei parametri di input
# When descrive quali fn: del modulo da testare
# Then contiene la descrizione delle asserzioni da svolgere

Feature: ecommerce

  # 0 Punti fedeltà
  Scenario: test_calcola_punti_fedelta_zero_punti_per_dieci_euro_acquisto
    Given un acquisto di 3 euro
    When un cliente conclude un acquisto fn:calcola_punti_fedelta ne calcola i punti fedeltà
    Then il cliente riceve 0 punti fedeltà perchè l'acquisto è inferiore a 10 euro

  # Prezzo scontato senza sconto
  Scenario: test_calcola_prezzo_scontato_senza_sconto
    Given un prezzo di 100 euro e uno sconto del 0 percento
    When fn:calcola_prezzo_scontato calcola il prezzo scontato
    Then il prezzo finale è 100 euro

  # Prezzo scontato con sconto
  Scenario: test_calcola_prezzo_scontato_con_sconto
    Given un prezzo di 100 euro e uno sconto del 20 percento
    When fn:calcola_prezzo_scontato calcola il prezzo scontato
    Then il prezzo finale è 80 euro

  # Prezzo scontato con sconto totale
  Scenario: test_calcola_prezzo_scontato_sconto_totale
    Given un prezzo di 100 euro e uno sconto del 100 percento
    When fn:calcola_prezzo_scontato calcola il prezzo scontato
    Then il prezzo finale è 0 euro

  # Prezzo negativo
  Scenario: test_calcola_prezzo_scontato_prezzo_negativo
    Given un prezzo di -10 euro e uno sconto del 20 percento
    When fn:calcola_prezzo_scontato calcola il prezzo scontato
    Then viene sollevata una ValueError

  # IVA con aliquota default
  Scenario: test_calcola_iva_con_aliquota_default
    Given un prezzo netto di 100 euro
    When fn:calcola_iva calcola il prezzo lordo
    Then il prezzo lordo è 122 euro

  # IVA con aliquota personalizzata
  Scenario: test_calcola_iva_con_aliquota_personalizzata
    Given un prezzo netto di 100 euro e un'aliquota del 10 percento
    When fn:calcola_iva calcola il prezzo lordo
    Then il prezzo lordo è 110 euro

  # IVA con prezzo negativo
  Scenario: test_calcola_iva_prezzo_negativo
    Given un prezzo netto di -10 euro
    When fn:calcola_iva calcola il prezzo lordo
    Then viene sollevata una ValueError

  # Creazione di una riga del carrello
  Scenario: test_crea_riga_carrello
    Given un prodotto "Mouse" con prezzo di 20 euro e quantità 2
    When fn:crea_riga_carrello crea una riga del carrello
    Then la riga contiene il nome "Mouse", il prezzo 20, la quantità 2 e il subtotale 40

  # Creazione riga con quantità non valida
  Scenario: test_crea_riga_carrello_quantita_non_valida
    Given un prodotto "Mouse" con prezzo di 20 euro e quantità 0
    When fn:crea_riga_carrello crea una riga del carrello
    Then viene sollevata una ValueError

  # Totale carrello vuoto
  Scenario: test_calcola_totale_carrello_vuoto
    Given un carrello vuoto
    When fn:calcola_totale_carrello calcola il totale
    Then il totale è 0 euro

  # Totale carrello con prodotti
  Scenario: test_calcola_totale_carrello_con_prodotti
    Given un carrello con due righe aventi subtotale di 20 e 30 euro
    When fn:calcola_totale_carrello calcola il totale
    Then il totale è 50 euro

  # Costo della spedizione
  Scenario: test_calcola_costo_spedizione

  # Costo della spedizione espressa
  Scenario: test_calcola_costo_spedizione_espressa

  # Costo della spedizione con peso negativo
  Scenario: test_calcola_costo_spedizione_peso_negativo

  # Punti fedeltà con 10 euro

  Scenario: test_calcola_punti_fedelta_con_moltiplicatore

  # Livello cliente bronze
  Scenario: test_promuovi_livello_cliente_bronze

  # Livello cliente silver
  Scenario: test_promuovi_livello_cliente_silver
    Given un cliente con 100 punti

  # Applicazione di un codice sconto valido
  Scenario: test_applica_codice_sconto_valido
    Given un prezzo di 100 euro e un codice "SCONTO20" associato a uno sconto del 20 percento
    When fn:applica_codice_sconto applica il codice "SCONTO20"
    Then il prezzo finale è 80 euro

  Scenario: test_applica_codice_sconto_inesistente
